cmake_minimum_required(VERSION 3.20)
# SPDX-License-Identifier: GPL-3.0-or-later
project(RadioModem VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RADIOMODEM_BUILD_TESTS "Build unit tests" ON)
add_library(radiomodem_core src/crc32c.cpp src/frame.cpp src/ofdm.cpp)
target_include_directories(radiomodem_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(radiomodem_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(radiomodem_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(radiomodem_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(radiomodem_demo src/main.cpp)
target_link_libraries(radiomodem_demo PRIVATE radiomodem_core)
if(RADIOMODEM_BUILD_TESTS)
  enable_testing()
  add_executable(radiomodem_tests tests/test_frame.cpp)
  target_link_libraries(radiomodem_tests PRIVATE radiomodem_core)
  add_test(NAME radiomodem_tests COMMAND radiomodem_tests)
  add_executable(radiomodem_ofdm_tests tests/test_ofdm.cpp)
  target_link_libraries(radiomodem_ofdm_tests PRIVATE radiomodem_core)
  add_test(NAME radiomodem_ofdm_tests COMMAND radiomodem_core)
endif()
